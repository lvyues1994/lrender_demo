cmake_minimum_required(VERSION 3.25)
project(lrender_demo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
# All options are explicit: no dependency tests, benchmarks, examples or downloads.
foreach(flag IN ITEMS LEXEC_BUILD_TESTS CO2_BUILD_TESTS CO2_BUILD_BENCHMARKS
    STACKFULL_BUILD_TESTS STACKFULL_BUILD_BENCH TC_BUILD_TESTS TC_BUILD_EXAMPLES TC_BUILD_BENCHMARKS
    LRENDER_BUILD_TESTS LRENDER_BUILD_EXAMPLES LRENDER_BUILD_PROBES LRENDER_KERNEL_RELOAD
    LQTEXEC_BUILD_TESTS LQTEXEC_BUILD_EXAMPLES LQTEXEC_BUILD_NETWORK LQTEXEC_INSTALL
    LQTEXEC_BUNDLE_LEXEC_HEADERS EXPECTED_BUILD_TESTS EXPECTED_BUILD_PACKAGE)
  set(${flag} OFF CACHE BOOL "Disabled for the demo dependency build" FORCE)
endforeach()
set(STACKFULL_EXCEPTIONS ON CACHE BOOL "Required by lrender" FORCE)
set(TC_BUILD_TRANSPILER ON CACHE BOOL "Build the kernel compiler" FORCE)
set(TC_ENABLE_VULKAN ON CACHE BOOL "Required for Vulkan verification" FORCE)
set(TC_ENABLE_OPENGL OFF CACHE BOOL "Not part of the first implementation" FORCE)
set(LQTEXEC_QT_MAJOR_VERSION 5 CACHE STRING "Use the available Qt 5.15 development environment")
if(NOT LQTEXEC_QT_MAJOR_VERSION STREQUAL "5")
  message(FATAL_ERROR "This initial build validates Qt 5.15 only; Qt 6 needs separate validation")
endif()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets)
include(cmake/LocalDependencies.cmake)
# Consumers must use the upstream WHOLE_ARCHIVE wrapper lr::render, not lr_render.
add_library(demo_core STATIC src/render/SessionState.cpp src/presentation/ReadbackAdapter.cpp src/render/FramePump.cpp)
target_include_directories(demo_core PUBLIC "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(demo_core PUBLIC Qt5::Widgets lqtexec::lqtexec lr::render lr::scene)
target_compile_features(demo_core PUBLIC cxx_std_17)
# Qt's `slots` macro corrupts stackfull's public `slots` data members.
# Self-owned Qt code must use Q_SIGNALS/Q_SLOTS/Q_EMIT instead of keywords.
target_compile_definitions(demo_core PUBLIC QT_NO_KEYWORDS)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(demo_core PUBLIC -fexceptions)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(demo_register target label)
  add_test(NAME ${target} COMMAND ${target})
  set_tests_properties(${target} PROPERTIES LABELS "${label}" TIMEOUT 60)
  if(NOT label STREQUAL "unit")
    set_tests_properties(${target} PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endfunction()
add_executable(demo_tests tests/main.cpp tests/link_contract_test.cpp tests/state_test.cpp tests/presentation_test.cpp tests/frame_pump_test.cpp)
target_include_directories(demo_tests PRIVATE "${PROJECT_SOURCE_DIR}/third_party/doctest")
target_link_libraries(demo_tests PRIVATE demo_core)
demo_register(demo_tests unit)
add_executable(demo_link_smoke tests/link_smoke.cpp)
target_link_libraries(demo_link_smoke PRIVATE demo_core)
demo_register(demo_link_smoke unit)
set_tests_properties(demo_tests demo_link_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME dependency_lock COMMAND "${Python3_EXECUTABLE}"
  "${PROJECT_SOURCE_DIR}/tests/check_dependency_lock.py" "${PROJECT_SOURCE_DIR}/cmake/VerifyDependencies.cmake")
set_tests_properties(dependency_lock PROPERTIES LABELS unit TIMEOUT 60)
add_test(NAME dependency_lock_sources COMMAND "${CMAKE_COMMAND}" -P "${PROJECT_SOURCE_DIR}/cmake/VerifyDependencies.cmake")
set_tests_properties(dependency_lock_sources PROPERTIES LABELS unit TIMEOUT 60)

# Public native platform boundary remains a separate self-owned target.
add_library(demo_native_surface STATIC src/presentation/NativeSurfaceFactory.cpp)
find_package(Vulkan REQUIRED)
find_path(DEMO_XCB_INCLUDE_DIR xcb/xcb.h REQUIRED)
find_library(DEMO_XCB_LIBRARY xcb REQUIRED)
target_include_directories(demo_native_surface PRIVATE "${DEMO_XCB_INCLUDE_DIR}")
target_link_libraries(demo_native_surface PUBLIC demo_core Vulkan::Vulkan PRIVATE "${DEMO_XCB_LIBRARY}")
target_sources(demo_tests PRIVATE tests/native_surface_policy_test.cpp)
target_link_libraries(demo_tests PRIVATE demo_native_surface)

if(NOT TARGET lr::present_vulkan)
  message(FATAL_ERROR "P2 requires the upstream Vulkan presenter target")
endif()
add_executable(demo_qt_vulkan_probe probes/qt_vulkan_probe.cpp)
target_link_libraries(demo_qt_vulkan_probe PRIVATE demo_native_surface lr::present_vulkan Vulkan::Vulkan)
demo_register(demo_qt_vulkan_probe platform)

# Strict sharing must never convert missing platform/GPU capability into a pass/skip.
option(DEMO_ENABLE_STRICT_SHARED_TEST "Register strict real-window Vulkan acceptance" OFF)
if(DEMO_ENABLE_STRICT_SHARED_TEST)
  add_test(NAME strict_shared COMMAND demo_qt_vulkan_probe --require-shared --frames 24)
  set_tests_properties(strict_shared PROPERTIES LABELS "platform;strict_shared" TIMEOUT 60)
endif()
